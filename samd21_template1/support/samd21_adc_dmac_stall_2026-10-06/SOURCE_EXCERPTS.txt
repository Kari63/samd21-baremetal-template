GCLK0 is configured from DFLL48M in Device_Startup/system_samd21.c.
The selected ADC clock is therefore nominally 48 MHz / 32 = 1.5 MHz.

3. ASYNCHRONOUS EVENT ROUTE

void evsys_route_tcc0_to_adc(void)
{
    evsys_channel_configure(EVSYS_TCC0_ADC_CHANNEL,
        EVSYS_ID_GEN_TCC0_MCX_0,
        EVSYS_CHANNEL_PATH_ASYNCHRONOUS);
    evsys_user_configure(EVSYS_ID_USER_ADC_START,
        EVSYS_TCC0_ADC_CHANNEL);
}

The current evsys_init() calls evsys_route_tcc0_to_adc(); the former
TC3-to-ADC route call is commented out.

4. UNLINKED ADC DMA DESCRIPTOR AND TRIGGER
Source: src/hal/dma/dma.cpp
Constants from include/hal/dma/dma.h:
    #define ADC_DMA_CHANNEL 1
    #define ADC_DMA_BUFFER_SAMPLES 16

Descriptor builder:

void dmac_build_channel6_adc_ping_pong(void)
{
    const uint16_t btctrl = DMAC_BTCTRL_VALID |
        DMAC_BTCTRL_DSTINC |
        DMAC_BTCTRL_BEATSIZE_HWORD |
        DMAC_BTCTRL_BLOCKACT_INT;

    descriptor_section[ADC_DMA_CHANNEL].btctrl = 0;
    descriptor_section[ADC_DMA_CHANNEL].btcnt = ADC_DMA_BUFFER_SAMPLES;
    descriptor_section[ADC_DMA_CHANNEL].srcaddr = (uint32_t)&ADC->RESULT.reg;
    descriptor_section[ADC_DMA_CHANNEL].dstaddr =
        (uint32_t)&adc_dma_buffers[adc_completed_buffer][ADC_DMA_BUFFER_SAMPLES];
    descriptor_section[ADC_DMA_CHANNEL].descaddr = 0;
    descriptor_section[ADC_DMA_CHANNEL].btctrl = btctrl;
    __DMB();
}

Channel setup:

void dmac_arm_channel6_adc_trigger(void)
{
    uint32_t primask = __get_PRIMASK();
    __disable_irq();

    DMAC->CHID.reg = DMAC_CHID_ID(ADC_DMA_CHANNEL);
    DMAC->CHCTRLA.reg = 0;
    DMAC->CHINTENSET.reg = DMAC_CHINTENSET_TCMPL | DMAC_CHINTENSET_TERR;
    DMAC->CHCTRLB.reg = DMAC_CHCTRLB_LVL(3) |
        DMAC_CHCTRLB_TRIGACT_BEAT |
        DMAC_CHCTRLB_TRIGSRC(ADC_DMAC_ID_RESRDY);
    DMAC->CHCTRLA.reg = DMAC_CHCTRLA_ENABLE;

    __set_PRIMASK(primask);
}

Block-complete handling (error path omitted here):

if (flags & DMAC_CHINTFLAG_TCMPL) {
    const uint8_t completed_buffer = adc_completed_buffer;
    uint32_t primask = __get_PRIMASK();
    __disable_irq();
    adc_completed_buffer ^= 1;
    dmac_build_channel6_adc_ping_pong();
    DMAC->CHID.reg = DMAC_CHID_ID(ADC_DMA_CHANNEL);
    DMAC->CHCTRLA.reg = DMAC_CHCTRLA_ENABLE;
    __set_PRIMASK(primask);
    adc_dma_samples_complete(adc_dma_buffers[completed_buffer],
        ADC_DMA_BUFFER_SAMPLES);
    adc_dma_block_count++;
}

Despite legacy function names containing "channel6" and "ping_pong", this
configuration selects ADC channel 1, has DESCADDR=0, and uses software A/B
buffer alternation. There is no linked ADC descriptor in this test.

5. TFT DRAW LOOP DISABLED FOR THE CURRENT RUN

void app_run(void)

The TFT driver and SERCOM5 are still initialized at startup. The continuous
app2() drawing workload, which starts TFT DMA channels 4/5, was not called.
